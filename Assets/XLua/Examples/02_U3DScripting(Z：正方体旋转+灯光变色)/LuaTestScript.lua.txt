--作用：实现让正方体旋转 和 改变灯光颜色
--思路：C#代码LuaBehavior来驱动这个Lua脚本， 得到
local speed = 10
local lightCpnt = nil

function start()
	print("lua start...")
	--Z：lightObject 是我们要挂载的物体的名称
	print("injected object", lightObject)
	lightCpnt= lightObject:GetComponent(typeof(CS.UnityEngine.Light))
end

function update()
	local r = CS.UnityEngine.Vector3.up * CS.UnityEngine.Time.deltaTime * speed
	--Z：Self是脚本挂载的物体
	self.transform:Rotate(r)
	lightCpnt.color = CS.UnityEngine.Color(CS.UnityEngine.Mathf.Sin(CS.UnityEngine.Time.time) / 2 + 0.5, 0, 0, 1)
end

function ondestroy()
    print("lua destroy")
end
